================================================================================
Recursive factorial
================================================================================

int factorial(int n) {
  if (n == 0)
    return 1;
  return n * factorial(n - 1);
}

================================================================================
Loop function
================================================================================

real my_sum(array[] real y) {
  real total = 0.0;
  for (i in 1:size(y)) {
    total += y[i];
  }
  return total;
}

================================================================================
Conditional function
================================================================================

real clamp(real x, real lo, real hi) {
  if (x < lo)
    return lo;
  else if (x > hi)
    return hi;
  return x;
}
