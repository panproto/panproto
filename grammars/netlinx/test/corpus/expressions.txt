// Basic logical word operations
result1 = x or y
result2 = a and b
result3 = c xor d
result4 = not e

// Complex logical word expressions
complex1 = (a or b) and not c
complex2 = not (x and y) or z
complex3 = a and b or c and d

// Mixed with comparison operators
comparison1 = (x > 5) or (y < 10)
comparison2 = (a == 1) and (b != 2)
comparison3 = not (value > threshold)

// Nested logic with parentheses
nested1 = ((a and b) or (c and d)) and not e
nested2 = not (a or not (b and c))

// Mixed with bitwise word operators
mixed1 = (a band $0F) or (b band $F0)
mixed2 = (flag1 and flag2) or (data1 bxor mask)

// With function calls
func_test1 = TIMELINE_ACTIVE(1) and not TIMELINE_ACTIVE(2)
func_test2 = LENGTH_ARRAY(arr1) > 0 or LENGTH_ARRAY(arr2) > 0

// Chained expressions
chained = a or b or c or d
chained2 = w and x and y and z
